Motors

⚙️ Motor Control (I²C)

# motor: 0=Left(M1), 1=Right(M2)
# direction: 0=forward, 1=backward
# speed: 0-255
def motor(m, direction, speed):
    buf = bytearray(3)
    buf[0] = 0x00 if m == 0 else 0x02
    buf[1] = direction
    buf[2] = speed
    i2c.write(0x10, buf)

👁 Sensors

LINE FOLLOWING (IR)
# Returns: 0 = black line, 1 = white surface
left  = pin13.read_digital()
right = pin14.read_digital()

if left == 0 and right == 1:
    motor(0, 0, 200); motor(1, 0, 0)  # veer left
elif left == 1 and right == 0:
    motor(0, 0, 0); motor(1, 0, 200)  # veer right
else:
    motor(0, 0, 200); motor(1, 0, 200)  # straight
ULTRASONIC (SR04 — pin1)
def distance_cm():
    pin1.write_digital(0); sleep_us(2)
    pin1.write_digital(1); sleep_us(10)
    pin1.write_digital(0)
    t = time_pulse_us(pin1, 1, 50000)
    return t / 58  # µs → cm

dist = distance_cm()
if dist < 20:
    display.show("!")   # obstacle!

Servos & Full Examples

🦾 Servo Motors

# Set servo angle via I2C (0–180 degrees)
# Register: 0x14 = S1, 0x15 = S2
def servo(which, angle):
    reg = 0x14 if which == 1 else 0x15
    i2c.write(0x10, bytearray([reg, angle]))

servo(1, 0)    # S1 to 0°
servo(1, 90)   # S1 to centre
servo(1, 180)  # S1 to 180°
servo(2, 45)   # S2 to 45°

# Slow sweep
for a in range(0, 181, 5):
    servo(1, a); sleep(30)
⚠ S1 is on P1 which is shared with the Ultrasonic Trig. Only use one at a time.

🚗 Obstacle Avoidance Example

from microbit import *
from machine import time_pulse_us
import random

i2c.init(freq=100000, sda=pin20, scl=pin19)

def motor(m, d, s):
    i2c.write(0x10, bytearray([0x00 if m==0 else 0x02, d, s]))

def dist():
    pin1.write_digital(0); sleep_us(2)
    pin1.write_digital(1); sleep_us(10)
    pin1.write_digital(0)
    return time_pulse_us(pin1, 1, 50000) / 58

while True:
    if dist() < 20:               # obstacle <20cm
        motor(0,0,0); motor(1,0,0)  # stop
        sleep(300)
        if random.randint(0,1):    # random turn
            motor(0,0,180); motor(1,1,180)
        else:
            motor(0,1,180); motor(1,0,180)
        sleep(500)
    else:
        motor(0,0,200); motor(1,0,200)

💬 Tips & Gotchas

Always call i2c.init() first
Motors won't respond without I²C initialized. Put it at the top of your program, before the main loop.
IR sensors: 0 = black, 1 = white
The IR sensors return 0 when they detect a dark/black surface. This is the opposite of what beginners expect!
NeoPixel: show() is mandatory
Setting np[0] = (255,0,0) buffers the colour. Nothing changes on the board until you call np.show().
Ultrasonic timeout
time_pulse_us() returns -1 if nothing is detected within 50ms. Always check for -1 before using the value.
